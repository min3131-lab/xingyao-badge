cmake_minimum_required(VERSION 3.20)
project(badge_desktop_preview LANGUAGES C CXX ASM)
get_filename_component(BADGE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${BADGE_ROOT}/managed_components/lvgl__lvgl" lvgl)
add_executable(preview preview.cc)
target_compile_features(preview PRIVATE cxx_std_17)
target_include_directories(preview PRIVATE "${BADGE_ROOT}/main/boards/waveshare/esp32-s3-touch-amoled-1.75")
target_link_libraries(preview PRIVATE lvgl)
